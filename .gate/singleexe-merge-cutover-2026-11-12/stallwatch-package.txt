== deployed v4 script (type from the laptop) ==
# tri7-stallwatch.ps1 — INTERIM reliability workaround for the embedded-node
# reconnect defect (node never re-dials after its last peer dies). Runs via
# scheduled task tri7gui-stallwatch (SYSTEM, every 10 min). If the wallet's
# node RPC is up AND the chain height is frozen across 75s AND every peer's
# last traffic is stale (>240s), restart the GUI once and log it. Removed
# when the reconnect dev item lands.
$log = 'C:\Coins\TRI-LAPTOP\stallwatch.log'
function Log($m) { ("[{0}] {1}" -f (Get-Date -Format o), $m) | Add-Content $log }
function RpcCall($b) {
  try { return (Invoke-RestMethod -Uri 'http://127.0.0.1:19114/' -Method Post -Body $b -ContentType 'application/json' -TimeoutSec 10).result }
  catch { return $null }
}
$gui = Get-Process -Name tri7_wallet_gui -ErrorAction SilentlyContinue
if (-not $gui) { Log "gui not running - nothing to watch"; exit }
$h1 = RpcCall '{"method":"getblockcount"}'
if ($null -eq $h1) {
  Log "rpc not answering - retrying once in 60s before considering a restart"
  Start-Sleep -Seconds 60
  $h1b = RpcCall '{"method":"getblockcount"}'
  if ($null -eq $h1b) {
    Log "rpc STILL down after retry -> restarting GUI (wedged)"
    taskkill /IM tri7_wallet_gui.exe /F 2>$null | Out-Null
    Start-Sleep -Seconds 4
    schtasks /Run /TN 'tri7gui-laptop' | Out-Null
    Log "restarted via tri7gui-laptop (wedged rpc)"
    exit
  }
  Log "rpc answered on retry - continuing health check"
  $h1 = $h1b
}
Start-Sleep -Seconds 75
$h2 = RpcCall '{"method":"getblockcount"}'
if ($null -eq $h2) { Log "rpc stopped answering mid-check - leaving it alone"; exit }
if ($h2 -ne $h1) { Log ("healthy: height {0} -> {1}" -f $h1, $h2); exit }
$stale = $false
$p = RpcCall '{"method":"getpeerinfo"}'
if ($null -eq $p -or $p.count -eq 0) { $stale = $true }
else {
  $peers = @($p.peers)
  $allStale = $true
  foreach ($peer in $peers) { if ($peer.last_traffic_ms_ago -le 240000) { $allStale = $false } }
  if ($allStale) { $stale = $true }
}
if (-not $stale) { Log ("height static {0} but peer traffic fresh - not stalling" -f $h2); exit }
Log ("STALL: height frozen at {0} with stale/dead peers -> restarting GUI" -f $h2)
taskkill /IM tri7_wallet_gui.exe /F 2>$null | Out-Null
Start-Sleep -Seconds 4
schtasks /Run /TN 'tri7gui-laptop' | Out-Null
Log "restarted via tri7gui-laptop"
