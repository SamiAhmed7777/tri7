            out_of_window, stg.base
        ));
    }
    let candidates_rel: Vec<tri7_core::stake::StakeCandidate> = candidates
        .iter()
        .filter(|c| (c.block_height as u64) >= stg.base)
        .map(|c| {

            r.block_height = stg.rel(c.block_height as u64);
            r
        })
        .collect();

        &stg.rel_index(),
        &candidates_rel,
        n_bits,
        fees,
        now,
        tri7_core::stake::MAX_STAKE_SEARCH_INTERVAL,
        balance,
        tip_height as usize + 1,
    )?;

    stake_st.coins_meeting_min_age = rep.coins_meeting_min_age;
    stake_st.searches += 1;

    };

        plan.signer.value,
        plan.coin_age / (tri7_core::consensus::COIN as u64 * 14_400), // coin-days at cap
        plan.reward,
        fees,
    ));

        .into_iter()
        .map(|(_h, tx)| tx)
        .collect();

        },
    )?;

            height_abs, e

                spent.push((vin.prev_hash, vin.prev_n));
            }
        }
    }
    pool.confirm_spend(&spent);

    write_state_atomic(cfg, &file).map_err(|e| format!("write state: {}", e))?;
    stake_st.last_stake = Some((
        height_abs,
        tri7_core::p2p::internal_to_display(&block.hash()),
        now_iso(),
    ));
    log(&format!(
        "STAKE MINTED → h{} {} (self-validated: kernel {}, nbits enforced) — relaying to {} peer(s)",
        height_abs,
        tri7_core::p2p::internal_to_display(&block.hash()),
        if outcome.kernel_enforced { "enforced" } else { "seeded" },
        peers.len(),
    ));
    Ok(true)
}

pub fn get_kernel_stake_modifier_with(
    idx: &ReplayIndex,
    block_from_height: usize,
    window: i64,
) -> Result<u64, String> {
    // Bounds-guard the entry index: callers stake from candidate heights that
    // must index `idx.entries` (window-relative in the follow-loop, absolute
    // in stake_mint). A foreign/out-of-range height must take the wallet.cpp
    // 2269-2271 skip-coin Err path, NOT panic the process (a panicked staking
    // loop killed the live daemon on 2026-11-02).
    let from = idx.entries.get(block_from_height).ok_or_else(|| {
        format!(
            "GetKernelStakeModifier: block height {} outside the index ({} entries) — coin skipped",
            block_from_height,
            idx.entries.len()
        )
    })?;
    let threshold = from.time as i64 + window;
    let mut p = block_from_height;
    let mut modifier_time = from.time as i64;
    loop {
        if modifier_time >= threshold {
            return Ok(idx.entries[p].stake_modifier);
        }
        if p + 1 >= idx.entries.len() {
            // kernel.cpp:256-268 — reached best chain end during replay.
            return Err(format!(
                "GetKernelStakeModifier: reached best block at height {} from block {}",
                p, block_from_height
            ));
        }
        p += 1;
        if idx.entries[p].generated_stake_modifier {
            modifier_time = idx.entries[p].time as i64;
        }
    }
}

pub fn append_block(&self, body: &[u8], height: Option<u64>) -> Result<Hash, String> {
        if body.len() < 80 {
            return Err(format!("append: body too short ({})", body.len()));
        }
        let inner = match self.inner() {
            Some(i) => i,
            None => {
                // Index still building: buffer the body, flush on ready. The
                // frame is durable in RAM only until then — same assumption
                // the recent-blocks cache makes — but NOTHING is lost to the
                // disk store once the build lands.
                let hash = header_hash(body)?;
                let mut p = self.pending.lock().unwrap();
                if p.iter().any(|(b, _)| header_hash(b).as_ref() == Ok(&hash)) {
                    return Ok(hash); // already buffered
                }
                p.push((body.to_vec(), height));
                return Ok(hash);
            }
        };
        // arrival-order guarantee: anything buffered during the build lands
        // before the first post-ready append
        self.flush_pending(&inner);
        let hash = header_hash(body)?;
        if self.block_by_hash(&hash)?.is_some() {
            return Ok(hash); // already on disk — nothing to do
        }
        if !inner.writable.load(Ordering::SeqCst) {
            return Err(
                "blocks file is not writable (append needs a tri7-owned blk file)".into(),
            );
        }
        let f = &inner.file;
        let off = f
            .metadata()
            .map_err(|e| format!("append: stat: {}", e))?
            .len();
        let mut frame = Vec::with_capacity(8 + body.len());
        frame.extend_from_slice(&FRAME_MAGIC.to_le_bytes());
        frame.extend_from_slice(&(body.len() as u32).to_le_bytes());
        frame.extend_from_slice(body);
        use std::io::Seek;
        let mut fh = f
            .try_clone()
            .map_err(|e| format!("append: clone handle: {}", e))?;
        fh.seek(std::io::SeekFrom::End(0))
            .map_err(|e| format!("append: seek: {}", e))?;
        fh.write_all(&frame)
            .map_err(|e| format!("append: write: {}", e))?;
        fh.sync_all().map_err(|e| format!("append: sync: {}", e))?;
        // index the frame
        let mut tx_keys = Vec::new();
        walk_tx_ranges(body, &mut |start, end| {
            tx_keys.push(key8(&sha256d(&body[start..end])));
            Ok(())
        })?;
        let seq = inner.blocks.frames as u32 + {
            let a = inner.appendix.read().unwrap();
            a.len() as u32
        };
        inner.appendix.write().unwrap().push(AppendedFrame {
            key: key8(&hash),
            off,
            seq,
            height,
            tx_keys,
        });
        Ok(hash)
    }
